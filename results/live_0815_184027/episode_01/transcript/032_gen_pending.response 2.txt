    def _generate_price_scenarios(self, current_hour, current_price, horizon=48):
        """Generate price scenarios using hour-of-day statistics and AR(1) process."""
        num_scenarios = 20
        scenarios = np.zeros((num_scenarios, horizon))
        
        for h in range(horizon):
            future_hour = (current_hour + h) % 24
            
            if future_hour in self.hour_price_dist:
                dist = self.hour_price_dist[future_hour]
                mean = dist.get('mean', 50.0)
                std = max(dist.get('std', 15.0), 5.0)
            else:
                mean = 50.0
                std = 15.0
            
            if h == 0:
                base = current_price
            else:
                base = scenarios[:, h-1]
            
            # AR(1): price_t = 0.7 * price_{t-1} + 0.3 * mean + noise
            mean_reversion = 0.7 * base + 0.3 * mean
            noise = np.random.normal(0, std, num_scenarios)
            scenarios[:, h] = np.maximum(mean_reversion + noise, 10.0)
        
        return scenarios

    def _generate_load_scenarios(self, current_hour, firm_load_mw, arriving_flex_mw, horizon=48):
        """Generate load scenarios based on hour-of-day patterns."""
        num_scenarios = 20
        scenarios = np.zeros((num_scenarios, horizon))
        
        for h in range(horizon):
            future_hour = (current_hour + h) % 24
            
            if future_hour in self.hour_load_dist:
                dist = self.hour_load_dist[future_hour]
                mean = dist.get('mean', 250.0)
                std = max(dist.get('std', 30.0), 5.0)
            else:
                mean = 250.0
                std = 30.0
            
            if h == 0:
                base = firm_load_mw
            else:
                base = scenarios[:, h-1]
            
            # Smooth load evolution
            mean_reversion = 0.8 * base + 0.2 * mean
            noise = np.random.normal(0, std, num_scenarios)
            scenarios[:, h] = np.maximum(mean_reversion + noise, 50.0)
        
        return scenarios

    def _solve_commitment_problem(
        self,
        current_hour,
        arriving_flex_mw,
        backlog_mwh,
        oldest_backlog_age_h,
        current_price,
        price_scenarios,
        load_scenarios,
    ):
        """Stage 1: Solve flex commitment problem over 6-hour horizon."""
        num_scenarios = price_scenarios.shape[0]
        commitment_h = self.commitment_horizon_h
        
        prices_commit = price_scenarios[:, :commitment_h]
        loads_commit = load_scenarios[:, :commitment_h]
        
        prices_mean = np.mean(prices_commit, axis=0)
        loads_mean = np.mean(loads_commit, axis=0)
        
        # Build cost coefficients
        serve_cost = np.zeros(commitment_h)
        for h in range(commitment_h):
            base_cost = prices_mean[h]
            
            # Grid stress factor
            firm_available = max(0, self.firm_base_capacity_mw - loads_mean[h])
            stress = max(0, 1.0 - (firm_available / self.flex_capacity_mw))
            
            serve_cost[h] = base_cost * (1.0 + stress)
        
        # Deadline penalty
        hours_to_deadline = self.backlog_deadline_h - oldest_backlog_age_h
        
        if hours_to_deadline <= 0:
            deferral_penalty = 500.0
            min_serve_total = backlog_mwh
        elif hours_to_deadline <= 2:
            deferral_penalty = np.mean(serve_cost) * 3.0
            min_serve_total = backlog_mwh * 0.8
        else:
            min_serve_rate = backlog_mwh / max(hours_to_deadline, 1.0)
            min_serve_total = min_serve_rate * commitment_h
            deferral_penalty = np.mean(serve_cost) * 0.5
        
        # LP formulation
        c = np.append(serve_cost, deferral_penalty)
        
        A_ub = []
        b_ub = []
        
        # Capacity constraints
        for h in range(commitment_h):
            row = np.zeros(commitment_h + 1)
            row[h] = 1.0
            A_ub.append(row)
            b_ub.append(self.flex_capacity_mw)
        
        # Non-negativity
        for h in range(commitment_h):
            row = np.zeros(commitment_h + 1)
            row[h] = -1.0
            A_ub.append(row)
            b_ub.append(0.0)
        
        row = np.zeros(commitment_h + 1)
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(0.0)
        
        # Total work constraint
        row = np.ones(commitment_h + 1) * -1.0
        A_ub.append(row)
        b_ub.append(-(arriving_flex_mw * commitment_h + backlog_mwh))
        
        # Deadline pacing
        if min_serve_total > 0:
            cumulative_must_serve = np.linspace(0, min_serve_total, commitment_h + 1)[1:]
            for h in range(commitment_h):
                row = np.zeros(commitment_h + 1)
                row[:h+1] = -1.0
                A_ub.append(row)
                b_ub.append(-cumulative_must_serve[h])
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        bounds = Bounds(np.zeros(commitment_h + 1), np.inf * np.ones(commitment_h + 1))
        
        try:
            x0 = np.append(
                np.full(commitment_h, min(arriving_flex_mw, self.flex_capacity_mw / 2.0)),
                max(0, backlog_mwh - arriving_flex_mw * commitment_h)
            )
            
            result = minimize(
                lambda x: np.dot(c, x),
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=LinearConstraint(A_ub, -np.inf, b_ub),
                options={'ftol': 1e-7, 'maxiter': 200}
            )
            
            if result.success:
                optimal_flex_serve = result.x[:commitment_h]
                cost = result.fun
            else:
                optimal_flex_serve = x0[:commitment_h]
                cost = np.dot(c, x0)
        except:
            optimal_flex_serve = np.full(commitment_h, min(arriving_flex_mw, 50.0))
            cost = np.dot(serve_cost, optimal_flex_serve) + deferral_penalty
        
        return optimal_flex_serve[0], serve_cost, cost

        else:
            price_p80 = np.max(prices_mean) if len(prices_mean) > 0 else 100.0
            price_p20 = np.min(prices_mean) if len(prices_mean) > 0 else 30.0
        
        price_range = max(price_p80 - price_p20, 1.0)
        current_normalized = (prices_mean[0] - price_p20) / price_range
        
        # Current hour dispatch decision
        current_load = firm_load_mw + flex_serve_mw
        
        # Look ahead for high loads
        if len(loads_mean) > 6:
            loads_ahead = loads_mean[1:7]
            max_load_ahead = np.max(loads_ahead)
        else:
            max_load_ahead = np.max(loads_mean) if len(loads_mean) > 0 else 300.0
        
        # Deadline pressure: reduce SOC target if backlog is aging
        hours_to_deadline = self.backlog_deadline_h - oldest_backlog_age_h
        if hours_to_deadline < 6:
            # Need to reserve capacity for forced serving
            target_soc = min(target_soc, 250.0)
        
        # Decision logic
        dispatch_mw = 0.0
        
        if battery_soc_mwh > max_soc * 0.95:
            dispatch_mw = -self.battery_power_rating_mw / 2.0
        elif battery_soc_mwh < min_soc * 1.1:
            dispatch_mw = self.battery_power_rating_mw / 2.0
        elif max_load_ahead > 320 and battery_soc_mwh > target_soc:
            dispatch_mw = -min(self.battery_power_rating_mw, (battery_soc_mwh - target_soc) / 2.0)
        elif current_normalized < 0.25 and battery_soc_mwh < target_soc:
            dispatch_mw = min(self.battery_power_rating_mw, (target_soc - battery_soc_mwh) / 2.0)
        elif current_normalized > 0.75 and battery_soc_mwh > target_soc:
            dispatch_mw = -min(self.battery_power_rating_mw, (battery_soc_mwh - min_soc) / 2.0)
        
        # Enforce SOC bounds
        if dispatch_mw > 0:
            max_charge = (max_soc - battery_soc_mwh)
            dispatch_mw = min(dispatch_mw, max_charge)
        else:
            max_discharge = (battery_soc_mwh - min_soc) * self.battery_efficiency
            dispatch_mw = max(dispatch_mw, -max_discharge)
        
        return np.clip(dispatch_mw, -self.battery_power_rating_mw, self.battery_power_rating_mw)

    def _update_distributions(self, hour_of_day, current_price, firm_load_mw):
        """Update hour-of-day statistics using Welford online algorithm."""
        
        if hour_of_day not in self.hour_price_dist:
            self.hour_price_dist[hour_of_day] = {
                'n': 0, 'mean': 0.0, 'M2': 0.0
            }
        
        dist_p = self.hour_price_dist[hour_of_day]
        dist_p['n'] += 1
        delta = current_price - dist_p['mean']
        dist_p['mean'] += delta / dist_p['n']
        dist_p['M2'] += delta * (current_price - dist_p['mean'])
        if dist_p['n'] > 1:
            dist_p['std'] = np.sqrt(dist_p['M2'] / dist_p['n'])
        else:
            dist_p['std'] = 15.0
        
        if hour_of_day not in self.hour_load_dist:
            self.hour_load_dist[hour_of_day] = {
                'n': 0, 'mean': 0.0, 'M2': 0.0
            }
        
        dist_l = self.hour_load_dist[hour_of_day]
        dist_l['n'] += 1
        delta = firm_load_mw - dist_l['mean']
        dist_l['mean'] += delta / dist_l['n']
        dist_l['M2'] += delta * (firm_load_mw - dist_l['mean'])
        if dist_l['n'] > 1:
            dist_l['std'] = np.sqrt(dist_l['M2'] / dist_l['n'])
        else:
            dist_l['std'] = 30.0
