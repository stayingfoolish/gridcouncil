    def _update_distributions(self, hour_of_day: int, current_price: float,
                             firm_load_mw: float):
        if hour_of_day not in self.hour_price_dist:
            self.hour_price_dist[hour_of_day] = {
                'n': 0, 'mean': 0.0, 'M2': 0.0, 'M3': 0.0, 'values': deque(maxlen=28)
            }
        
        dist = self.hour_price_dist[hour_of_day]
        dist['n'] += 1
        dist['values'].append(current_price)
        delta = current_price - dist['mean']
        dist['mean'] += delta / dist['n']
        delta2 = current_price - dist['mean']
        dist['M2'] += delta * delta2
        dist['M3'] += delta * delta * delta2
        
        if dist['n'] >= 4:
            dist['std'] = np.sqrt(dist['M2'] / (dist['n'] - 1))
            dist['skew'] = dist['M3'] / (dist['n'] - 1) / (max(dist['std'], 0.1) ** 3) if dist['std'] > 0 else 0.0
            dist['p25'] = np.percentile(list(dist['values']), 25)
            dist['p75'] = np.percentile(list(dist['values']), 75)
            dist['p90'] = np.percentile(list(dist['values']), 90)
        else:
            dist['std'] = max(current_price * 0.25, 5.0)
            dist['skew'] = 0.5
            dist['p25'] = current_price
            dist['p75'] = current_price
            dist['p90'] = current_price
        
        if hour_of_day not in self.hour_load_dist:
            self.hour_load_dist[hour_of_day] = {
                'n': 0, 'mean': 0.0, 'M2': 0.0, 'values': deque(maxlen=28)
            }
        
        dist = self.hour_load_dist[hour_of_day]
        dist['n'] += 1
        dist['values'].append(firm_load_mw)
        delta = firm_load_mw - dist['mean']
        dist['mean'] += delta / dist['n']
        delta2 = firm_load_mw - dist['mean']
        dist['M2'] += delta * delta2
        
        if dist['n'] >= 4:
            dist['std'] = np.sqrt(dist['M2'] / (dist['n'] - 1))
            dist['p95'] = np.percentile(list(dist['values']), 95)
        else:
            dist['std'] = max(firm_load_mw * 0.15, 20.0)
            dist['p95'] = firm_load_mw

    def _forecast_scenarios(self, current_hour: int, num_scenarios: int = 50) -> tuple:
        price_scenarios = np.zeros((num_scenarios, 48))
        load_scenarios = np.zeros((num_scenarios, 48))
        
        for h in range(48):
            future_hour = (current_hour + h) % 24
            
            if future_hour in self.hour_price_dist:
                p_stats = self.hour_price_dist[future_hour]
                mean_p = p_stats['mean']
                std_p = p_stats.get('std', mean_p * 0.25)
                skew_p = p_stats.get('skew', 0.3)
                
                z = np.random.normal(0, 1, num_scenarios)
                price_scenarios[:, h] = mean_p + std_p * (z + skew_p * (z**2 - 1) / 6)
                price_scenarios[:, h] = np.maximum(price_scenarios[:, h], 5.0)
            else:
                price_scenarios[:, h] = np.random.normal(50, 20, num_scenarios)
            
            if future_hour in self.hour_load_dist:
                l_stats = self.hour_load_dist[future_hour]
                mean_l = l_stats['mean']
                std_l = l_stats.get('std', mean_l * 0.15)
                
                load_scenarios[:, h] = np.random.normal(mean_l, std_l, num_scenarios)
                load_scenarios[:, h] = np.clip(load_scenarios[:, h], 150, 350)
            else:
                load_scenarios[:, h] = np.random.normal(250, 40, num_scenarios)
        
        return price_scenarios, load_scenarios

    def _compute_deadline_risk(self, backlog_mwh: float, oldest_backlog_age_h: float,
                              load_scenarios: np.ndarray) -> float:
        if backlog_mwh < 1.0 or oldest_backlog_age_h >= 24.0:
            return 0.0
        
        hours_remaining = int(self.backlog_deadline_h - oldest_backlog_age_h)
        if hours_remaining <= 0:
            return 1.0
        
        num_scenarios = load_scenarios.shape[0]
        cumulative_firm = np.zeros(num_scenarios)
        
        for h in range(min(hours_remaining, 24)):
            loads_h = load_scenarios[:, h]
            firm_available = np.maximum(0, self.firm_base_capacity_mw - loads_h)
            firm_available = np.minimum(firm_available, self.flex_capacity_mw)
            cumulative_firm += firm_available
        
        risk = np.mean(cumulative_firm < backlog_mwh)
        return float(risk)

    def _compute_serve_cost_profile(self, price_scenarios: np.ndarray,
                                   load_scenarios: np.ndarray) -> np.ndarray:
        num_hours = price_scenarios.shape[1]
        serve_costs = np.zeros(num_hours)
        
        prices_mean = np.mean(price_scenarios, axis=0)
        loads_mean = np.mean(load_scenarios, axis=0)
        
        for h in range(num_hours):
            firm_available = max(0, self.firm_base_capacity_mw - loads_mean[h])
            firm_available = min(firm_available, self.flex_capacity_mw)
            
            capacity_tightness = 1.0 - (firm_available / self.flex_capacity_mw)
            opportunity_cost = 2.0 * capacity_tightness * prices_mean[h]
            
            serve_costs[h] = prices_mean[h] + opportunity_cost
        
        return serve_costs

    def _compute_battery_target_trajectory(self, current_hour: int,
                                          backlog_mwh: float,
                                          oldest_backlog_age_h: float,
                                          battery_soc_mwh: float,
                                          price_scenarios: np.ndarray) -> np.ndarray:
        min_soc = self.soc_safety_margin_pct * self.battery_capacity_mwh
        max_soc = 0.95 * self.battery_capacity_mwh
        
        horizon = 24
        target_soc = np.zeros(horizon)
        
        prices_mean = np.mean(price_scenarios[:, :horizon], axis=0)
        prices_std = np.std(price_scenarios[:, :horizon], axis=0)
        
        price_baseline = np.mean(prices_mean)
        
        for h in range(horizon):
            price_normalized = (prices_mean[h] - price_baseline) / max(price_baseline, 1.0)
            
            if price_normalized > 0.3:
                target_soc[h] = min_soc + 0.2 * (max_soc - min_soc)
            elif price_normalized < -0.3:
                target_soc[h] = min_soc + 0.8 * (max_soc - min_soc)
            else:
                target_soc[h] = min_soc + 0.5 * (max_soc - min_soc)
        
        deadline_risk = self._compute_deadline_risk(backlog_mwh, oldest_backlog_age_h,
                                                    price_scenarios)
        if deadline_risk > 0.7:
            for h in range(horizon):
                target_soc[h] = min_soc + 0.3 * (max_soc - min_soc)
        
        target_soc = np.clip(target_soc, min_soc, max_soc)
        
        return target_soc

            serve_fraction = 0.6

        current_cost_ratio = current_price / max(min_serve_cost, 1.0)

            serve_fraction *= 0.5
        elif current_cost_ratio < 0.8:
            serve_fraction *= 1.5
        
        backlog_to_serve = min(backlog_mwh * serve_fraction, self.flex_capacity_mw - arriving_flex_mw)
        backlog_to_serve = max(0, backlog_to_serve)
        
        total_flex_serve = arriving_flex_mw + backlog_to_serve
        total_flex_serve = min(total_flex_serve, self.flex_capacity_mw)
        
        return max(0, total_flex_serve)
