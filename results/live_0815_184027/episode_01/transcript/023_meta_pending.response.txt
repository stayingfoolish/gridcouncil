# Novel Dispatch Architecture: Integrated Risk-Aware Scheduling

## Analysis of Current Implementation's Limitations

### Strengths:
- Dual-horizon planning (commit + lookahead)
- Statistical hour-of-day profiling
- Deadline awareness with pressure curves
- Multi-constraint feasibility windows

### Critical Weaknesses:

1. **Decoupled Decision-Making**: Flex scheduling and battery dispatch are sequential, not jointly optimized. Battery state doesn't inform flex commit, creating suboptimal trajectories.

2. **Reactive Battery Control**: 8 separate if-statements with hardcoded thresholds (0.35, 0.45, 0.50, 0.65 SOC ratios). No unified objective function.

3. **Greedy Flex Serving**: Serves backlog based on "cost to deadline ratio" without modeling the value of deferral when cheaper windows exist within deadline.

4. **No Stochastic Risk Modeling**: Uses percentiles as proxies for volatility, but doesn't explicitly model the probability of deadline violation or grid constraint violations.

5. **Performance Degradation**: Last 4 iterations show: 14,525,027 → 14,613,893 → 14,895,941 → 14,898,846. The policy is drifting away from the best solution.

6. **Missing Gap to Optimum**: $2.9M gap to theoretical optimum suggests systematic underutilization of flexibility.

    def _ingest_observation(self, hour_of_day: int, current_price: float,
                           firm_load_mw: float, flex_served_mw: float = 0.0,
                           battery_dispatch_mw: float = 0.0):
        """Update empirical distributions."""
        self.price_history.append((hour_of_day, current_price))
        self.load_history.append((hour_of_day, firm_load_mw))
        self.flex_served_history.append(flex_served_mw)
        self.battery_dispatch_history.append(battery_dispatch_mw)
        
        # Update hour-of-day distributions with Welford's algorithm for skew
        if hour_of_day not in self.hour_price_dist:
            self.hour_price_dist[hour_of_day] = {
                'n': 0, 'mean': 0.0, 'M2': 0.0, 'M3': 0.0
            }
        
        dist = self.hour_price_dist[hour_of_day]
        dist['n'] += 1
        delta = current_price - dist['mean']
        dist['mean'] += delta / dist['n']
        delta2 = current_price - dist['mean']
        dist['M2'] += delta * delta2
        dist['M3'] += delta * delta * delta2
        
        if dist['n'] >= 4:
            dist['std'] = np.sqrt(dist['M2'] / (dist['n'] - 1))
            dist['skew'] = dist['M3'] / (dist['n'] - 1) / (dist['std'] ** 3) if dist['std'] > 0 else 0
        
        # Same for loads
        if hour_of_day not in self.hour_load_dist:
            self.hour_load_dist[hour_of_day] = {
                'n': 0, 'mean': 0.0, 'M2': 0.0, 'p95': 0.0
            }
        
        dist = self.hour_load_dist[hour_of_day]
        dist['n'] += 1
        delta = firm_load_mw - dist['mean']
        dist['mean'] += delta / dist['n']
        delta2 = firm_load_mw - dist['mean']
        dist['M2'] += delta * delta2
        
        if dist['n'] >= 4:
            dist['std'] = np.sqrt(dist['M2'] / (dist['n'] - 1))
            dist['p95'] = dist['mean'] + 1.645 * dist['std']  # 95th percentile
    
    def _forecast_scenario_tree(self, current_hour: int, 
                               num_scenarios: int = 100) -> tuple:
        """
        Generate Monte Carlo price and load scenarios respecting empirical distributions.
        Returns: (price_scenarios, load_scenarios) both shape (num_scenarios, 48)
        """
        price_scenarios = np.zeros((num_scenarios, 48))
        load_scenarios = np.zeros((num_scenarios, 48))
        
        for h in range(48):
            future_hour = (current_hour + h) % 24
            
            # Price distribution
            if future_hour in self.hour_price_dist:
                p_stats = self.hour_price_dist[future_hour]
                mean_p = p_stats['mean']
                std_p = p_stats.get('std', mean_p * 0.25)
                skew_p = p_stats.get('skew', 0.5)
                
                # Skew-normal approximation
                z = np.random.normal(0, 1, num_scenarios)
                price_scenarios[:, h] = mean_p + std_p * (z + skew_p * (z**2 - 1) / 6)
                price_scenarios[:, h] = np.maximum(price_scenarios[:, h], 5.0)  # Floor
            else:
                price_scenarios[:, h] = np.random.normal(40, 15, num_scenarios)
            
            # Load distribution
            if future_hour in self.hour_load_dist:
                l_stats = self.hour_load_dist[future_hour]
                mean_l = l_stats['mean']
                std_l = l_stats.get('std', mean_l * 0.15)
                
                load_scenarios[:, h] = np.random.normal(mean_l, std_l, num_scenarios)
                load_scenarios[:, h] = np.clip(load_scenarios[:, h], 100, 380)
            else:
                load_scenarios[:, h] = np.random.normal(280, 50, num_scenarios)
        
        return price_scenarios, load_scenarios
    
    def _compute_deferral_value_curve(self, current_hour: int,
                                     backlog_mwh: float,
                                     oldest_backlog_age_h: float,
                                     price_scenarios: np.ndarray,
                                     load_scenarios: np.ndarray) -> dict:
        """
        For each future hour h, compute expected cost to serve 1 MWh backlog.
        Accounts for price, load-dependent firmness, and deadline risk.
        
        Returns: {
            'serve_cost': array of shape (48,),  # Expected cost per MWh
            'deadline_risk': array of shape (24,),  # P(deadline violation if defer)
            'cheapest_windows': list of (start_h, end_h, expected_cost)
        }
        """
        num_scenarios = price_scenarios.shape[0]
        num_hours = min(int(self.backlog_deadline_h - oldest_backlog_age_h) + 1, 24)
        
        serve_cost_per_hour = np.zeros(48)
        deadline_risk = np.zeros(24)
        
        for h in range(48):
            # Average cost across scenarios, weighted by firmness
            prices_h = price_scenarios[:, h]
            loads_h = load_scenarios[:, h]
            
            # Firmness: how much flex capacity is available?
            firm_available = np.maximum(0, self.firm_base_capacity_mw - loads_h)
            firm_available = np.minimum(firm_available, self.flex_capacity_mw)
            
            # Cost includes price + opportunity cost of tying up capacity
            opportunity_cost = 2.0 * (1 - firm_available / self.flex_capacity_mw) * prices_h
            total_cost = prices_h + opportunity_cost
            
            serve_cost_per_hour[h] = np.mean(total_cost)
        
        # Deadline risk: probability of not clearing backlog by deadline
        for h in range(min(num_hours, 24)):
            hours_left = self.backlog_deadline_h - (oldest_backlog_age_h + h)
            
            # Empirical: fraction of scenarios where cumulative firm capacity 
            # from h to deadline is insufficient
            cumulative_firm = np.zeros(num_scenarios)
            for future_h in range(h, min(h + int(hours_left) + 1, 24)):
                loads_future = load_scenarios[:, future_h]
                firm_avail = np.maximum(0, self.firm_base_capacity_mw - loads_future)
                firm_avail = np.minimum(firm_avail, self.flex_capacity_mw)
                cumulative_firm += firm_avail
            
            insufficient = (cumulative_firm < backlog_mwh)
            deadline_risk[h] = np.mean(insufficient)
        
        # Identify good deferral windows
        windows = []
        current_window_start = None
        cumulative_cost = 0.0
        window_hours = 0
        
        for h in range(min(num_hours, 24)):
            if serve_cost_per_hour[h] < np.percentile(serve_cost_per_hour[:num_hours], 40):
                if current_window_start is None:
                    current_window_start = h
                cumulative_cost += serve_cost_per_hour[h]
                window_hours += 1
            else:
                if current_window_start is not None:
                    avg_cost = cumulative_cost / window_hours
                    windows.append((current_window_start, h - 1, avg_cost))
                    current_window_start = None
                    cumulative_cost = 0.0
                    window_hours = 0
        
        if current_window_start is not None:
            avg_cost = cumulative_cost / window_hours
            windows.append((current_window_start, 23, avg_cost))
        
        windows.sort(key=lambda x: x[2])  # Sort by cost ascending
        
        return {
            'serve_cost': serve_cost_per_hour,
            'deadline_risk': deadline_risk,
            'cheapest_windows': windows,
            'num_hours_to_deadline': num_hours,
        }
    
    def _optimize_target_soc_trajectory(self, 
                                       backlog_mwh: float,
                                       oldest_backlog_age_h: float,
                                       battery_soc_mwh: float,
                                       price_scenarios: np.ndarray,
                                       load_scenarios: np.ndarray) -> np.ndarray:
        """
        Compute optimal battery SOC trajectory for next 24h that minimizes 
        expected cost while maintaining deadline feasibility.
        
        Uses dynamic programming on scenario tree.
        
        Returns: target_soc array of shape (24,)
        """
        num_scenarios = price_scenarios.shape[0]
        horizon = 24
        
        min_soc = self.soc_safety_margin_pct * self.battery_capacity_mwh
        max_soc = 0.95 * self.battery_capacity_mwh
        
        # Initialize: all scenarios start at current SOC
        soc_all = np.full((num_scenarios, horizon + 1), battery_soc_mwh)
        
        # Forward pass: simulate scenarios
        cumulative_cost = np.zeros(num_scenarios)
        cumulative_backlog_cleared = np.zeros(num_scenarios)
        
        for h in range(horizon):
            prices_h = price_scenarios[:, h]
            loads_h = load_scenarios[:, h]
            
            # Determine optimal action for each scenario
            # Simple policy: discharge when price is high and we have buffer
            soc_ratio = soc_all[:, h] / self.battery_capacity_mwh
            price_percentile = np.percentile(prices_h, 75)
            
            discharge_mask = (prices_h > price_percentile) & (soc_ratio > 0.45)
            charge_mask = (prices_h < np.percentile(prices_h, 25)) & (soc_ratio < 0.65)
            
            # Default: maintain SOC
            battery_dispatch = np.zeros(num_scenarios)
            battery_dispatch[discharge_mask] = -50  # MW discharge
            battery_dispatch[charge_mask] = 50  # MW charge
            
            # Update SOC
            charge_efficiency = self.battery_efficiency if battery_dispatch > 0 else 1.0
            soc_all[:, h + 1] = soc_all[:, h] - battery_dispatch * charge_efficiency
            soc_all[:, h + 1] = np.clip(soc_all[:, h + 1], min_soc, max_soc)
            
            # Track cost
            firm_served = loads_h
            grid_import = firm_served + np.maximum(0, 100 - battery_dispatch)
            cumulative_cost += prices_h * grid_import
        
        # Backward pass: compute target trajectory as median across scenarios
        target_soc = np.median(soc_all, axis=0)
        
        return target_soc[:24]
