class Policy:
    def __init__(self):
        """Initializes the policy with dynamic price-aware thresholds."""
        self.base_charge_threshold = 0.25      # Lower floor
        self.base_discharge_threshold = 0.75   # Higher ceiling
        self.price_sensitivity = 0.15          # Volatility multiplier
        self.recent_prices = []                # Track last 5 price pairs
        
    def _calculate_adaptive_thresholds(self, current_grid_buy_price, current_grid_sell_price):
        """Adjust thresholds based on current price levels relative to recent history."""
        if not self.recent_prices:
            return self.base_discharge_threshold, self.base_charge_threshold
        
        recent_buys = [p[0] for p in self.recent_prices[-5:]]
        recent_sells = [p[1] for p in self.recent_prices[-5:]]
        
        avg_buy = sum(recent_buys) / len(recent_buys)
        avg_sell = sum(recent_sells) / len(recent_sells)
        
        # If buy price is LOW relative to average, willing to charge at HIGHER SoC
        buy_ratio = current_grid_buy_price / avg_buy if avg_buy > 0 else 1.0
        discharge_threshold = self.base_discharge_threshold + (1.0 - buy_ratio) * self.price_sensitivity
        
        # If sell price is HIGH relative to average, willing to discharge at LOWER SoC
        sell_ratio = current_grid_sell_price / avg_sell if avg_sell > 0 else 1.0
        charge_threshold = self.base_charge_threshold - (sell_ratio - 1.0) * self.price_sensitivity
        
        return (
            min(0.95, max(0.55, discharge_threshold)),  # Bounds: [55%, 95%]
            max(0.05, min(0.40, charge_threshold))      # Bounds: [5%, 40%]
        )
    
    def take_action(self,
        current_energy_stored_kwh: float,
        current_pv_generation_kw: float,
        current_demand_kw: float,
        current_grid_buy_price: float,
        current_grid_sell_price: float,
        battery_capacity_kwh: float,
    ):
        """Determines the target action with price-adaptive thresholds."""
        
        # Track price history
        self.recent_prices.append((current_grid_buy_price, current_grid_sell_price))
        if len(self.recent_prices) > 5:
            self.recent_prices.pop(0)
        
        # Calculate adaptive thresholds
        discharge_threshold, charge_threshold = self._calculate_adaptive_thresholds(
            current_grid_buy_price, current_grid_sell_price
        )
        
        energy_stored_ratio = current_energy_stored_kwh / battery_capacity_kwh if battery_capacity_kwh > 0 else 0
        net_generation = current_pv_generation_kw - current_demand_kw
        
        # [Rest of original logic remains the same, but uses dynamic thresholds]
        # Example application:
        if net_generation > 0:
            if energy_stored_ratio < discharge_threshold:  # NOW DYNAMIC
                max_charge = min(10, max(0, net_generation), 
                               battery_capacity_kwh - current_energy_stored_kwh)
                if max_charge > 0.1:
                    action_kw = max_charge
                    reason = f"Charging {max_charge:.2f}kW (threshold: {discharge_threshold*100:.0f}%, SoC: {energy_stored_ratio*100:.1f}%)"
                else:
                    action_kw = 0
                    reason = "Battery full"
            else:
                action_kw = 0
                reason = f"Battery above adaptive threshold ({discharge_threshold*100:.0f}%), selling surplus"
        
        elif (current_demand_kw - current_pv_generation_kw) > 0:
            if energy_stored_ratio > charge_threshold:  # NOW DYNAMIC
                energy_deficit = current_demand_kw - current_pv_generation_kw
                max_discharge = min(5, energy_deficit, current_energy_stored_kwh)
                action_kw = -max_discharge if max_discharge > 0.1 else 0
                reason = f"Discharging {max_discharge:.2f}kW (threshold: {charge_threshold*100:.0f}%, SoC: {energy_stored_ratio*100:.1f}%)"
            else:
                action_kw = 0
                reason = f"Battery below adaptive threshold ({charge_threshold*100:.0f}%), buying from grid"
        else:
            action_kw = 0
            reason = "Supply balanced with demand"
        
        action_kw = max(-5, min(10, action_kw))
        return action_kw, reason
