# Fundamental Rethink: From Margin-Chasing to Cost Minimization

## Critical Analysis of Current Implementation

**Current Strengths:**
- Multi-horizon predictions and regime detection
- Adaptive thresholds based on volatility
- Emergency handling for extreme SOC

**Fatal Limitations:**
1. **Misaligned Objective**: Chasing percentage margins (6-12%) regardless of absolute cost - this is profit-seeking, not cost minimization
2. **Delayed Decision Making**: Waits for high confidence signals → misses low-cost periods
3. **SOC Oscillation**: Gets stuck between 40-60% (0% utilization) - no strategic energy positioning
4. **Regime Blindness**: Same logic across all regimes instead of regime-specific strategies
5. **No Demand Coupling**: Ignores that we should charge when demand is LOW and prices are LOW simultaneously

---

## Novel Approach: Predictive Cost Minimization with Demand Synchronization

### Core Philosophy
**Rather than "buy low, sell high," focus on: "Pre-position battery to minimize total system cost (grid purchases)"**

```python
from collections import deque
import math
from dataclasses import dataclass
from typing import Tuple, List
import numpy as np

@dataclass
class CostRegime:
    cost_percentile: float  # Where is current price in historical distribution
    demand_forecast: float  # Expected demand in next horizon
    cheapest_time_ahead: int  # Steps until next low-cost period
    cost_efficiency: float  # Can we serve upcoming demand cheaper with battery?

class ImprovedPolicy:
    def __init__(self):
        self.max_charge_power = 10
        self.max_discharge_power = 5
        self.battery_capacity = 50
        
        # Core: Cost-tracking (not margin-tracking)
        self.price_history = deque(maxlen=168)  # 1 week of hourly data
        self.demand_history = deque(maxlen=168)
        self.cost_paid_history = deque(maxlen=168)
        
        # New: Regime-specific SOC targets
        self.regime_soc_targets = {
            'cheap_period': 0.95,      # Fill when price is low
            'expensive_period': 0.25,  # Deplete when price is high
            'rising_trend': 0.80,      # Pre-position before price rise
            'falling_trend': 0.30,     # Empty before price fall
            'volatile': 0.60           # Stay neutral in chaos
        }
        
        # New: Cost thresholds (not margin thresholds)
        self.cost_percentile_thresholds = {
            'charge_trigger': 0.30,    # Charge when price is in bottom 30%
            'discharge_trigger': 0.75  # Discharge when price is in top 25%
        }
        
        self.step_counter = 0
        self.current_action = "IDLE"
        self.action_entry_time = 0
        
        # Learning: track actual cost avoided
        self.cost_savings_log = deque(maxlen=100)
        self.action_outcomes = deque(maxlen=50)

    def _calculate_cost_percentile(self, current_price: float) -> float:
        """Where does current price rank in recent history?"""
        if len(self.price_history) < 20:
            return 0.5
        
        prices = list(self.price_history)
        sorted_prices = sorted(prices)
        
        # Find percentile rank
        rank = sum(1 for p in sorted_prices if p <= current_price) / len(sorted_prices)
        return rank

    def _forecast_demand_pattern(self) -> dict:
        """Simple: is demand currently HIGH or LOW relative to history?"""
        if len(self.demand_history) < 24:
            return {'demand_level': 'unknown', 'forecast_avg': 0, 'is_peak': False}
        
        recent_24h = list(self.demand_history)[-24:]
        current_demand = recent_24h[-1] if recent_24h else 0
        avg_demand_24h = sum(recent_24h) / len(recent_24h)
        
        is_peak = current_demand > avg_demand_24h * 1.2
        demand_level = 'peak' if is_peak else 'normal'
        
        return {
            'demand_level': demand_level,
            'forecast_avg': avg_demand_24h,
            'is_peak': is_peak,
            'current': current_demand
        }

    def _identify_cost_regime(self, current_price: float, demand_info: dict) -> str:
        """Classify market state to determine optimal SOC target"""
        if len(self.price_history) < 20:
            return 'volatile'
        
        recent = list(self.price_history)[-20:]
        price_slope = (recent[-1] - recent[0]) / (recent[0] + 1e-6)
        price_volatility = np.std(recent) / (np.mean(recent) + 1e-6)
        
        cost_percentile = self._calculate_cost_percentile(current_price)
        
        # Decision tree for regimes
        if cost_percentile < 0.25:
            return 'cheap_period'
        elif cost_percentile > 0.75:
            return 'expensive_period'
        elif price_slope > 0.03:
            return 'rising_trend'
        elif price_slope < -0.03:
            return 'falling_trend'
        else:
            return 'volatile'

    def _calculate_cost_efficiency(self,
        current_price: float,
        future_demand: float,
        future_prices: List[float],
        current_soc: float
    ) -> float:
        """
        Can we save money by using battery for upcoming demand?
        Returns: Net savings if we fully deploy battery for future demand
        """
        if future_demand <= 0:
            return 0
        
        battery_available = current_soc * self.battery_capacity
        
        # Cost if we use grid only
        grid_cost = future_demand * np.mean(future_prices)
        
        # Cost if we use battery + grid
        battery_used = min(battery_available, future_demand)
        grid_used = max(0, future_demand - battery_used)
        
        # Opportunity cost: what could we sell this battery for later?
        battery_opportunity_cost = battery_used * max(future_prices) if max(future_prices) > current_price else 0
        
        # Net savings
        savings = (grid_cost - 
                  (grid_used * np.mean(future_prices) + battery_opportunity_cost))
        
        return savings

    def _should_charge(self,
        current_price: float,
        regime: str,
        current_soc: float,
        max_available: float,
        demand_forecast: dict,
        future_prices: List[float]
    ) -> Tuple[bool, float]:
        """
        Decide if we should charge based on:
        1. Is current price in bottom 30%?
        2. Is SOC below regime target?
        3. Do we have charging capacity?
        """
        
        if max_available < 0.5:
            return False, 0.0
        
        cost_percentile = self._calculate_cost_percentile(current_price)
        target_soc = self.regime_soc_targets.get(regime, 0.5)
        
        # Don't overcharge in any regime
        if current_soc > 0.90:
            return False, 0.0
        
        # Primary trigger: price is cheap
        if cost_percentile < self.cost_percentile_thresholds['charge_trigger']:
            power = min(max_available, self.max_charge_power)
            return True, power
        
        # Secondary trigger: strategic pre-positioning
        # If price is falling and we're below target, charge harder
        if len(self.price_history) >= 3:
            recent_slope = (list(self.price_history)[-1] - list(self.price_history)[-3]) / 3
            if recent_slope < 0 and current_soc < target_soc * 1.1:
                power = min(max_available * 0.7, self.max_charge_power * 0.6)
                return True, power
        
        return False, 0.0

    def _should_discharge(self,
        current_price: float,
        regime: str,
        current_soc: float,
        max_available: float,
        demand_info: dict,
        future_prices: List[float]
    ) -> Tuple[bool, float]:
        """
        Decide if we should discharge based on:
        1. Is current price in top 25%?
        2. Is SOC above regime target?
        3. Can we meet demand with battery?
        """
        
        if max_available < 0.5:
            return False, 0.0
        
        cost_percentile = self._calculate_cost_percentile(current_price)
        target_soc = self.regime_soc_targets.get(regime, 0.5)
        
        # Don't overdeplete in any regime
        if current_soc < 0.15:
            return False, 0.0
        
        # Primary trigger: price is expensive
        if cost_percentile > self.cost_percentile_thresholds['discharge_trigger']:
            power = min(max_available, self.max_discharge_power)
            return True, power
        
        # Secondary trigger: peak demand + high SOC
        if demand_info['is_peak'] and current_soc > target_soc:
            # Reduce grid draw during peak
            power = min(max_available * 0.5, self.max_discharge_power * 0.5)
            return True, power
        
        # Tertiary trigger: prepare for rising prices
        if len(self.price_history) >= 3:
            recent_slope = (list(self.price_history)[-1] - list(self.price_history)[-3]) / 3
            if recent_slope > 0 and current_soc > target_soc * 0.9:
                power = min(max_available * 0.6, self.max_discharge_power * 0.7)
                return True, power
        
        return False, 0.0

    def take_action(self,
        current_energy_stored_kwh: float,
        current_pv_generation_kw: float,
        current_demand_kw: float,
        current_grid_buy_price: float,
        current_grid_sell_price: float,
        battery_capacity_kwh: float,
    ) -> float:
        
        self.battery_capacity = battery_capacity_kwh
        self.price_history.append(current_grid_buy_price)
        self.demand_history.append(current_demand_kw)
        self.step_counter += 1
        
        soc = current_energy_stored_kwh / battery_capacity_kwh
        
        energy_to_full = battery_capacity_kwh - current_energy_stored_kwh
        max_charge_available = min(self.max_charge_power, energy_to_full)
        max_discharge_available = min(self.max_discharge_power, current_energy_stored_kwh)
        
        # Get context
        regime = self._identify_cost_regime(current_grid_buy_price, {})
        demand_info = self._forecast_demand_pattern()
        
        # Simple 24-step ahead forecast
        future_prices = self._simple_forecast(24) if len(self.price_history) >= 5 else [current_grid_buy_price] * 24
        
        # Decision logic: COST-BASED, not margin-based
        should_charge, charge_power = self._should_charge(
            current_grid_buy_price,
            regime,
            soc,
            max_charge_available,
            demand_info,
            future_prices
        )
        
        should_discharge, discharge_power = self._should_discharge(
            current_grid_sell_price,
            regime,
            soc,
            max_discharge_available,
            demand_info,
            future_prices
        )
        
        # Priority: discharge takes precedence (serve load at high prices)
        if should_discharge:
            return -discharge_power
        
        # Then charge (fill when cheap)
        if should_charge:
            return charge_power
        
        # No action
        return 0.0

    def _simple_forecast(self, horizon: int) -> List[float]:
        """Simple exponential smoothing forecast"""
        if len(self.price_history) < 3:
            return [list(self.price_history)[-1] if self.price_history else 50] * horizon
        
        prices = list(self.price_history)
        alpha = 0.3
        forecast = []
        current = prices[-1]
        trend = (prices[-1] - prices[-3]) / 2
        
        for _ in range(horizon):
            current = alpha * current + (1 - alpha) * (current + trend * 0.8)
            current = max(10, min(100, current))  # Reasonable bounds
            forecast.append(current)
        
        return forecast
```
